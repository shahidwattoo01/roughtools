# ─── Database Backup Schedule ──────────────────────────────
# Add these to your production server crontab:
#
# crontab -e
#
# Daily backup at 3 AM UTC
# Weekly full backup on Sundays at 2 AM UTC
# Keep 7 daily backups, 4 weekly backups
# Upload to S3 after each backup (via s3cmd/rclone)
# S3 bucket: roughtools-backups
#
# ─── Daily incremental ─────────────────────────────────────
0 3 * * * docker exec -u root roughtools-db-1 pg_dump -U postgres -Fc roughtools | gzip > /backups/daily/roughtools-$(date +\%Y\%m\%d).sql.gz && \
s3cmd put /backups/daily/roughtools-$(date +\%Y\%m\%d).sql.gz s3://roughtools-backups/daily/ && \
find /backups/daily/ -name "*.sql.gz" -mtime +7 -delete

# ─── Weekly full backup ───────────────────────────────────
0 2 * * 0 docker exec -u root roughtools-db-1 pg_dump -U postgres -Fc roughtools | gzip > /backups/weekly/roughtools-$(date +\%Y\%m\%d).sql.gz && \
s3cmd put /backups/weekly/roughtools-$(date +\%Y\%m\%d).sql.gz s3://roughtools-backups/weekly/ && \
find /backups/weekly/ -name "*.sql.gz" -mtime +30 -delete

# ─── Verify backups ──────────────────────────────────────
15 6 * * * echo "SELECT count(*) FROM pg_stat_activity;" | docker exec -i roughtools-db-1 psql -U postgres -d roughtools -tAc
